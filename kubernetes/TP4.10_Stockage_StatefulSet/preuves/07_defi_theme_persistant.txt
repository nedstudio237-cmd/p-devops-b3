# Installation d'un thème (Astra) directement dans wp-content/themes (stocké sur le PVC wordpress-html)
$ kubectl -n wordpress exec deploy/wordpress -- sh -c 'cd /tmp && curl -sfLo astra.zip https://downloads.wordpress.org/theme/astra.zip && php -r "\$z=new ZipArchive; \$z->open(\"astra.zip\"); \$z->extractTo(\"/var/www/html/wp-content/themes/\"); echo \"extrait\n\";" && chown -R www-data: /var/www/html/wp-content/themes/astra'
extrait

$ kubectl -n wordpress exec deploy/wordpress -- ls /var/www/html/wp-content/themes

index.php
